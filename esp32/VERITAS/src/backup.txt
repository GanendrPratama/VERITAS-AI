#include <Arduino.h>
#include <Wire.h>
#include <MAX30105.h>

// ESP32-C3 SuperMini Pins
#define SDA_PIN 5
#define SCL_PIN 6

MAX30105 sensor;

void setup() {
  setCpuFrequencyMhz(80);
  Serial.begin(115200);
  delay(2000); // Wait for USB CDC Serial to attach

  Serial.println("\n==========================================");
  Serial.println("   MAX30102 DIRECT HARDWARE POWER TEST    ");
  Serial.println("==========================================");

  // 1. Initialize I2C Bus
  Serial.printf("[1] Initializing I2C (SDA=GPIO%d, SCL=GPIO%d)...\n", SDA_PIN, SCL_PIN);
  Wire.begin(SDA_PIN, SCL_PIN);
  Wire.setClock(100000); // Standard 100kHz speed for unstable connections

  // 2. Direct I2C Probe
  Serial.println("[2] Probing address 0x57 directly...");
  Wire.beginTransmission(0x57);
  byte error = Wire.endTransmission();

  if (error != 0) {
    Serial.printf("❌ FAIL: No answer on 0x57 (I2C Error code: %d)\n", error);
    Serial.println("   --> SDA/SCL wires are not making electrical contact!");
    return;
  }
  Serial.println("✅ SUCCESS: Chip responded on address 0x57!");

  // 3. Initialize Sensor Library
  Serial.println("[3] Initializing MAX30102 registers...");
  if (!sensor.begin(Wire, I2C_SPEED_STANDARD)) {
    Serial.println("❌ FAIL: Library failed to read Chip ID!");
    return;
  }

  // 4. Force LEDs to MAXIMUM BRIGHTNESS
  Serial.println("[4] FORCING RED AND IR LEDs TO 100% POWER...");
  
  // Power configuration: 
  // Red = 0xFF (Max current ~50mA)
  // IR = 0xFF (Max current ~50mA)
  sensor.setup(0x1F, 1, 2, 400, 411, 4096); 
  sensor.setPulseAmplitudeRed(0xFF); // MAX RED
  sensor.setPulseAmplitudeIR(0xFF);  // MAX IR
  sensor.setPulseAmplitudeGreen(0x00);

  Serial.println("\n🔥 LEDs SHOULD BE GLOWING BRIGHT RED RIGHT NOW! 🔥");
}

void loop() {
  // If I2C works, continuously read raw IR value
  long irValue = sensor.getIR();
  long redValue = sensor.getRed();

  if (irValue == 0 && redValue == 0) {
    Serial.println("[READ ERROR] Bus disconnected mid-read!");
  } else {
    Serial.printf("RAW DATA -> Red: %ld | IR: %ld\n", redValue, irValue);
  }

  delay(500);
}
